description: "Build a working React Native component that detects jailbreaks, root, and mock locations using every JailMonkey API method with TypeScript."
---

This guide walks you through importing JailMonkey, calling both synchronous and asynchronous detection methods, and building a complete React Native component. By the end you will have a working screen that displays real-time detection results for every JailMonkey API.

<Steps>
  <Step title="Import JailMonkey">
    Add the default import at the top of your component file. JailMonkey exports a single object with all detection methods.

    ```typescript
    import JailMonkey from 'jail-monkey';
    ```
  </Step>

  <Step title="Call synchronous methods">
    Most JailMonkey methods are synchronous and can be called directly in your render logic or anywhere in your component body.

    ```typescript
    import JailMonkey from 'jail-monkey';

    // Cross-platform (iOS and Android)
    const isJailBroken = JailMonkey.isJailBroken();       // boolean
    const canMockLocation = JailMonkey.canMockLocation();  // boolean
    const trustFall = JailMonkey.trustFall();              // boolean (either of the above is true)

    // iOS only (returns false on Android)
    const message = JailMonkey.jailBrokenMessage();        // string — reason for detection

    // Android only (returns false on iOS)
    const hookDetected = JailMonkey.hookDetected();
    const isOnExternalStorage = JailMonkey.isOnExternalStorage();
    const adbEnabled = JailMonkey.AdbEnabled();
    ```
  </Step>

  <Step title="Handle async methods with useEffect">
    Two methods — `isDebuggedMode` and `isDevelopmentSettingsMode` — return Promises. Use `useState` and `useEffect` to resolve them and store the result.

    ```typescript
    import { useState, useEffect } from 'react';
    import JailMonkey from 'jail-monkey';

    export default function SecurityScreen() {
      const [isDebuggedMode, setIsDebuggedMode] = useState<boolean | undefined>();
      const [isDevelopmentSettingsMode, setIsDevelopmentSettingsMode] = useState<
        boolean | undefined
      >();

      useEffect(() => {
        JailMonkey.isDebuggedMode()
          .then(setIsDebuggedMode)
          .catch(console.warn);
      }, []);

      useEffect(() => {
        JailMonkey.isDevelopmentSettingsMode()
          .then(setIsDevelopmentSettingsMode)
          .catch(console.warn);
      }, []);

      // ...
    }
    ```

    <Note>
      `isDevelopmentSettingsMode` is Android-only. On iOS it resolves to `false`. You can still call it safely on both platforms — the Promise will always resolve.
    </Note>
  </Step>

  <Step title="Build the full detection component">
    Here is a complete working component based on the JailMonkey example app. It renders all detection results for both platforms.

    ```typescript App.tsx
    import * as React from 'react';
    import { SafeAreaView, View, StyleSheet, Text } from 'react-native';
    import JailMonkey from 'jail-monkey';

    const { useState, useEffect } = React;

    export default function App() {
      const [isDevelopmentSettingsMode, setIsDevelopmentSettingsMode] = useState<
        boolean | undefined
      >();
      const [isDebuggedMode, setIsDebuggedMode] = useState<boolean | undefined>();

      useEffect(() => {
        JailMonkey.isDevelopmentSettingsMode()
          .then(setIsDevelopmentSettingsMode)
          .catch(console.warn);
      }, []);

      useEffect(() => {
        JailMonkey.isDebuggedMode()
          .then(setIsDebuggedMode)
          .catch(console.warn);
      }, []);

      return (
        <SafeAreaView style={styles.root}>
          <View style={styles.content}>
            <Text style={styles.title}>Android & iOS</Text>
            <Row label="isJailBroken" value={JailMonkey.isJailBroken()} />
            <Row label="canMockLocation" value={JailMonkey.canMockLocation()} />
            <Row label="trustFall" value={JailMonkey.trustFall()} />
            <Row label="isDebuggedMode" value={isDebuggedMode} />

            <Text style={styles.title}>Android only</Text>
            <Text style={styles.note}>These APIs always return false on iOS.</Text>
            <Row label="hookDetected" value={JailMonkey.hookDetected()} />
            <Row label="isOnExternalStorage" value={JailMonkey.isOnExternalStorage()} />
            <Row label="AdbEnabled" value={JailMonkey.AdbEnabled()} />
            <Row label="isDevelopmentSettingsMode" value={isDevelopmentSettingsMode} />
          </View>
        </SafeAreaView>
      );
    }

    function Row({
      label,
      value,
    }: {
      label: string;
      value: string | boolean | undefined;
    }) {
      return (
        <View style={styles.row}>
          <Text style={styles.label}>{label}:</Text>
          <Text style={styles.value}>{value?.toString() ?? 'unknown'}</Text>
        </View>
      );
    }

    const styles = StyleSheet.create({
      root: { flex: 1, backgroundColor: '#fff' },
      content: { padding: 20 },
      title: { fontSize: 26, fontWeight: '700', marginTop: 20, marginBottom: 5 },
      note: { fontSize: 11, color: '#888', marginBottom: 10 },
      row: { flexDirection: 'row', marginBottom: 5 },
      label: { fontSize: 16, fontWeight: '700', marginRight: 5 },
      value: { fontSize: 16, color: '#444' },
    });
    ```
  </Step>

  <Step title="Gate features based on detection results">
    Use the detection results to restrict access, log a security event, or prompt the user. The `trustFall()` method is a convenient shorthand that returns `true` if either `isJailBroken()` or `canMockLocation()` is `true`.

    ```typescript
    import JailMonkey from 'jail-monkey';

    function checkDeviceTrust() {
      if (JailMonkey.trustFall()) {
        // Device is jailbroken/rooted or can mock location
        return false;
      }
      return true;
    }

    // Or check each condition individually for more granular control
    function getDeviceSecurityStatus() {
      return {
        jailBroken: JailMonkey.isJailBroken(),
        canFakeLocation: JailMonkey.canMockLocation(),
        // iOS only — empty string means no jailbreak detected
        jailBreakReason: JailMonkey.jailBrokenMessage(),
      };
    }
    ```
  </Step>
</Steps>

<Tip>
  During development, your app is likely running on an emulator or simulator. These environments are typically rooted, so JailMonkey's detection methods will return `true` even on a clean dev machine. Add an environment guard (for example, check `__DEV__`) to skip security checks during development and avoid constant false positives.

  ```typescript
  if (!__DEV__ && JailMonkey.isJailBroken()) {
    // Only enforce in production builds
  }
  ```
</Tip>
